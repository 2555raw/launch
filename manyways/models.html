Read the quickstart
quickstart.sh 
# one endpoint, every model in the register
curl https://api.hanamy.dev/v1/chat/completions \
  -H "Authorization: Bearer $MW_CLIENT_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "model": "claude-opus-5",
    "messages": [{"role":"user","content":"say hi"}],
    "stream": true
  }'
